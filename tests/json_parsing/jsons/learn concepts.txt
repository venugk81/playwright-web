# moderate_students.json
[
  {
    "id": 101, "name": "Alice",
    "scores": {"math": 88, "science": 75, "english": 92},
    "subjects": ["Math", "Science", "English"],
    "passed": true
  },
  {
    "id": 102, "name": "Bob",
    "scores": {"math": 55, "science": 60, "english": 48},
    "subjects": ["Math", "Science"],
    "passed": false
  }
]

Read nested nodes
for student in students:
    name = student["name"]
    math = student["scores"]["math"]

    # Loop nested dict
    for subj, score in \
        student["scores"].items():
        print(f"{subj}: {score}")

    # Loop subjects list
    for subj in student["subjects"]:
        print(subj)

 Compute & add new fields
for student in students:
    scores = student["scores"]

    # Calculate average
    avg = sum(scores.values()) \
          / len(scores)
    student["average"] = round(avg, 2)

    # Add a rank field
    student["rank"] = (
        "Distinction" if avg >= 80
        else "Pass" if avg >= 50
        else "Fail"
    )

dit nodes based on condition
# Condition: if any score < 50, mark "passed" as False and flag weak subjects
for student in students:
    weak = [subj for subj, score in
            student["scores"].items() if score < 50]

    if weak:
        student["passed"] = False
        student["weak_subjects"] = weak
        print(f"{student['name']} failed: {weak}")

# Remove a field from all records
for student in students:
    student.pop("subjects", None)  # safe delete

# complex_students.json
{
  "institution": "Sunrise University",
  "students": [
    {
      "id": 201, "name": "Alice",
      "profile": {
        "email": "alice@uni.edu",
        "address": {"city": "Hyderabad", "state": "TS"}
      },
      "semesters": [
        {
          "term": "Fall 2024",
          "courses": [
            {"code": "CS101", "score": 88, "credits": 4},
            {"code": "MA201", "score": 45, "credits": 3}
          ]
        }
      ],
      "extracurricular": ["Chess Club", "Robotics"]
    }
  ]
}

Read deeply nested nodes
with open("complex_students.json") as f:
    data = json.load(f)

inst = data["institution"]
students = data["students"]

for s in students:
    city = s["profile"]["address"]["city"]
    email = s["profile"].get("email")

    for sem in s["semesters"]:
      term = sem["term"]
      for c in sem["courses"]:
        print(c["code"], c["score"])


Compute GPA across semesters
for s in students:
  total_pts = 0
  total_cr  = 0

  for sem in s["semesters"]:
    for c in sem["courses"]:
      total_pts += c["score"] * c["credits"]
      total_cr  += c["credits"]

  gpa = round(total_pts / total_cr, 2)
  s["gpa"] = gpa  # add new field
  print(f"{s['name']}: GPA {gpa}")

Read deeply nested nodes
with open("complex_students.json") as f:
    data = json.load(f)

inst = data["institution"]
students = data["students"]

for s in students:
    city = s["profile"]["address"]["city"]
    email = s["profile"].get("email")

    for sem in s["semesters"]:
      term = sem["term"]
      for c in sem["courses"]:
        print(c["code"], c["score"])

Compute GPA across semesters
for s in students:
  total_pts = 0
  total_cr  = 0

  for sem in s["semesters"]:
    for c in sem["courses"]:
      total_pts += c["score"] * c["credits"]
      total_cr  += c["credits"]

  gpa = round(total_pts / total_cr, 2)
  s["gpa"] = gpa  # add new field
  print(f"{s['name']}: GPA {gpa}")

Edit nested nodes based on condition
# Condition: flag each course with score < 50 as "remedial": True
# and add a "backlog_courses" list to the semester
for s in students:
    for sem in s["semesters"]:
        backlogs = []
        for c in sem["courses"]:
            if c["score"] < 50:
                c["remedial"] = True       # edit the course node
                backlogs.append(c["code"])
        sem["backlog_courses"] = backlogs  # add to semester

# Condition: drop extracurricular if GPA < 60 (academic probation)
for s in students:
    if s.get("gpa", 100) < 60:
        s["extracurricular"] = []
        s["status"] = "Academic Probation"

# Save everything back
with open("complex_students.json", "w") as f:
    json.dump(data, f, indent=2)
